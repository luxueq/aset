```go
// internal/provider/local/adapter.go
type LocalAdapter struct {
    openaiAdapter *openai.OpenAIAdapter

}

func (a *LocalAdapter) ConvertRequest(ctx context.Context, req *CanonicalRequest) (*http.Request, error) {
    endpoint := a.endpoints[stripPrefix(req.Model)]
    // Override base URL to local endpoint
    httpReq, _ := http.NewRequestWithContext(ctx, "POST", endpoint+"/v1/chat/completions", body)
    // No auth header needed for local models (or configurable)
    return httpReq, nil
}
```

```go
func MetricsMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        recorder := &responseRecorder{ResponseWriter: w}
        next.ServeHTTP(recorder, r)
        duration := time.Since(start)

        requestDuration.WithLabelValues(r.Method, r.URL.Path, strconv.Itoa(recorder.StatusCode)).Observe(duration.Seconds())
        requestTotal.WithLabelValues(r.Method, r.URL.Path, strconv.Itoa(recorder.StatusCode)).Inc()
    })
}
```

### Task 6.4: Request Logging Middleware (with Sanitization)

**Type:** execute

**Action:**
Implement request/response logging middleware with configurable sanitization.

```go
// internal/observability/logging.go
type SanitizeConfig struct {
    Fields      []string `yaml:"fields"`       // 要脱敏的字段
    MaskPattern string   `yaml:"mask_pattern"` // 默认: "{prefix}***"
}

func LoggingMiddleware(logger *zap.Logger, cfg SanitizeConfig) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            start := time.Now()
            recorder := &responseRecorder{ResponseWriter: w}

            // Log request (sanitized)
            logger.Info("request_started",
                zap.String("method", r.Method),
                zap.String("path", r.URL.Path),
                zap.String("api_key_prefix", sanitizeKey(extractKey(r))),
            )

            next.ServeHTTP(recorder, r)

            // Log response
            logger.Info("request_completed",
                zap.String("method", r.Method),
                zap.String("path", r.URL.Path),
                zap.Int("status", recorder.StatusCode),
                zap.Duration("duration", time.Since(start)),
                zap.String("model", recorder.Model),
                zap.Int64("prompt_tokens", recorder.PromptTokens),
                zap.Int64("completion_tokens", recorder.CompletionTokens),
                zap.String("cache", recorder.CacheStatus),
            )
        })
    }
}

func sanitizeKey(key string) string {
    if len(key) <= 8 {
        return "***"
    }
    return key[:8] + "***"
}
```

DEBUG level: log full request/response body (when enabled).

**read_first:**
- `.planning/phases/01-core-gateway/01-CONTEXT.md` §日志与脱敏 (D-34~D-37)

**acceptance_criteria:**
- INFO level: API key shown as prefix + `***` (first 8 chars)
- INFO level: messages content NOT logged (only token count)
- INFO level: status code, duration, model, tokens, cache status logged
- DEBUG level (when enabled): full request/response body logged
- Sanitization fields configurable via config
- No sensitive data (full API keys, user messages) in INFO logs

### Task 6.5: Dockerfile + docker-compose for Development

**Type:** execute

**Action:**
Create Dockerfile and docker-compose.yml for local development with PostgreSQL + Redis.

```dockerfile
# Dockerfile
FROM golang:1.22-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 go build -o /aigateway ./cmd/aigateway

FROM alpine:3.19
COPY --from=builder /aigateway /aigateway
COPY config/ /etc/aigateway/
EXPOSE 8080
ENTRYPOINT ["/aigateway", "--config", "/etc/aigateway/default.yaml"]
```

```yaml
# docker-compose.yml
services:
  aigateway:
    build: .
    ports: ["8080:8080"]
    depends_on: [postgres, redis]
    environment:
      - AIGATEWAY_DATABASE_URL=postgres://aigw:aigw@postgres:5432/aigateway
      - AIGATEWAY_REDIS_URL=redis://redis:6379

  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: aigateway
      POSTGRES_USER: aigw
      POSTGRES_PASSWORD: aigw
    volumes: ["pgdata:/var/lib/postgresql/data"]

  redis:
    image: redis:7-alpine
    volumes: ["redisdata:/data"]
```
