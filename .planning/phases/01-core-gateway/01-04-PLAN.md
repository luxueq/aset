```go
// internal/ratelimit/local.go
type LocalBucket struct {
    buckets sync.Map // key -> *bucket
    mu      sync.RWMutex
}

type bucket struct {
    tokens    float64
    lastTime  time.Time
    capacity  float64
    refillRate float64 // tokens per second
}

func (lb *LocalBucket) Allow(key string, tokens int) bool {
    b := lb.getOrCreate(key)
    b.refill()
    if b.tokens >= float64(tokens) {
        b.tokens -= float64(tokens)
        return true
    }
    return false
}
```

### Task 4.2: L2 Redis Rate Limiter with Lua Script

**Type:** execute

**Action:**
Implement Redis-backed distributed rate limiter using Lua scripts for atomicity.

```lua
-- Rate limit check with sliding window
local key = KEYS[1]
local limit = tonumber(ARGV[1])
local window = tonumber(ARGV[2])
local now = tonumber(ARGV[3])
local tokens = tonumber(ARGV[4])

-- Remove expired entries
redis.call('ZREMRANGEBYSCORE', key, 0, now - window)

-- Count current
local current = redis.call('ZCARD', key)

if current + tokens <= limit then
    redis.call('ZADD', key, now, now .. '-' .. math.random(1000000))
    redis.call('PEXPIRE', key, window)
    return 1  -- allowed
end
return 0  -- denied
```

### Task 4.3: 4-Level Rate Limit Middleware

**Type:** execute

**Action:**
Implement middleware that checks all 4 rate limit dimensions: Per API Key, Per Tenant, Per Model, Global.

```go
func RateLimitMiddleware(limiter *CompositeLimiter) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            key := extractAPIKey(r)
            tenant := extractTenant(r)
            model := extractModel(r)
            tokens := estimateTokens(r)

            // Check all 4 levels: L1 first, then L2
            checks := []struct{
                dimension string
                limitKey  string
            }{
                {"global", "global"},
                {"model", "model:" + model},
                {"tenant", "tenant:" + tenant},
                {"key", "key:" + key},
            }

            for _, c := range checks {
                allowed, info, err := limiter.Allow(r.Context(), c.limitKey, tokens)
                if !allowed {
                    w.Header().Set("X-RateLimit-Limit", strconv.Itoa(info.Limit))
                    w.Header().Set("X-RateLimit-Remaining", "0")
                    w.Header().Set("X-RateLimit-Reset", strconv.FormatInt(info.Reset, 10))
                    w.Header().Set("Retry-After", strconv.Itoa(int(info.RetryAfter)))
                    writeError(w, http.StatusTooManyRequests, "rate_limit_exceeded")
                    return
                }
            }

            // Set rate limit headers on successful response
            next.ServeHTTP(w, r)
        })
    }
}
```
